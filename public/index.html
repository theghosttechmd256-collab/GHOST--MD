require("dotenv").config(); const express = require("express"); const pino = require("pino"); const QRCode = require("qrcode"); const fs = require("node:fs"); const path = require("node:path"); const { default: makeWASocket, useMultiFileAuthState, fetchLatestBaileysVersion, DisconnectReason } = require("@whiskeysockets/baileys"); const { handleMessage, commandCount } = require("./handler"); const app = express(); const logger = pino({ level: process.env.LOG_LEVEL || "info" }); const PORT = Number(process.env.PORT || 3000); const AUTH_DIR = process.env.AUTH_DIR || path.join(process.cwd(), "auth_info"); const state = { sock: null, qr: null, connected: false, starting: false, pairing: false, startPromise: null, lastError: null }; app.set("trust proxy", 1); app.disable("x-powered-by"); app.use(express.json({ limit: "12kb" })); app.use(express.static(path.join(__dirname, "..", "public"))); app.use( "/assets", express.static(path.join(__dirname, "..", "assets"), { maxAge: "7d" }) ); // Security headers app.use((_req, res, next) => { res.setHeader("X-Content-Type-Options", "nosniff"); res.setHeader("Referrer-Policy", "no-referrer"); res.setHeader("Cache-Control", "no-store"); next(); }); // Basic per-IP request limiter const requestLimits = new Map(); function rateLimit(maxRequests, windowMs) { return (req, res, next) => { const ip = req.ip || req.socket.remoteAddress || "unknown"; const now = Date.now(); const record = requestLimits.get(ip); if (record && now < record.reset && record.count >= maxRequests) { return res.status(429).json({ error: "Too many requests. Please wait and try again." }); } if (!record || now >= record.reset) { requestLimits.set(ip, { count: 1, reset: now + windowMs }); } else { record.count++; } next(); }; } const publicLimit = rateLimit(30, 10 * 60 * 1000); const pairLimit = rateLimit(5, 10 * 60 * 1000); async function start() { if (state.connected) return; if (state.startPromise) return state.startPromise; state.starting = true; state.startPromise = (async () => { fs.mkdirSync(AUTH_DIR, { recursive: true }); const { state: auth, saveCreds } = await useMultiFileAuthState(AUTH_DIR); const { version } = await fetchLatestBaileysVersion(); const sock = makeWASocket({ version, auth, logger, printQRInTerminal: false, browser: ["GHOST-MD", "Chrome", "2.5.0"], markOnlineOnConnect: false, syncFullHistory: false }); state.sock = sock; sock.ev.on("creds.update", saveCreds); sock.ev.on("connection.update", async ({ connection, qr, lastDisconnect }) => { if (qr) { try { state.qr = await QRCode.toDataURL(qr); } catch (err) { logger.error(err); } } if (connection === "open") { state.connected = true; state.starting = false; state.qr = null; state.lastError = null; logger.info("GHOST-MD connected"); } if (connection === "close") { state.connected = false; state.starting = false; state.qr = null; state.sock = null; const code = lastDisconnect?.error?.output?.statusCode; if (code !== DisconnectReason.loggedOut) { logger.warn("Connection closed; reconnecting in 5 seconds"); setTimeout(() => { start().catch(err => logger.error(err)); }, 5000); } else { state.lastError = "Logged out. Pair your WhatsApp again."; } } }); sock.ev.on("messages.upsert", async ({ messages, type }) => { if (type !== "notify") return; for (const msg of messages) { try { await handleMessage(sock, msg); } catch (err) { logger.error(err); } } }); })(); try { await state.startPromise; } catch (err) { state.lastError = "WhatsApp startup failed. Check server logs."; state.starting = false; state.sock = null; logger.error(err); throw err; } finally { state.startPromise = null; } } // Homepage app.get("/", (_req, res) => res.redirect("/index.html")); // Fix QR and Pair page links app.get("/qr", (_req, res) => res.redirect("/qr.html")); app.get("/pair", (_req, res) => res.redirect("/pair.html")); // Health check app.get("/health", (_req, res) => { res.json({ ok: true, bot: "GHOST-MD", version: "2.5.0", commands: commandCount }); }); // Public connection status app.get("/api/status", publicLimit, (_req, res) => { res.json({ connected: state.connected, starting: state.starting, qrReady: Boolean(state.qr), lastError: state.lastError }); }); // Start WhatsApp connection app.post("/api/start", publicLimit, async (_req, res) => { try { await start(); res.json({ ok: true, message: "Connection starting. Try again shortly." }); } catch (err) { res.status(503).json({ error: state.lastError || "Could not start connection." }); } }); // QR endpoint app.get("/api/qr", publicLimit, async (_req, res) => { if (state.connected) { return res.status(409).json({ error: "WhatsApp is already connected." }); } try { if (!state.sock) await start(); } catch (err) { return res.status(503).json({ error: state.lastError || "Connection startup failed." }); } if (!state.qr) { return res.status(202).json({ waiting: true, message: "QR is being generated. Refresh shortly." }); } res.json({ qr: state.qr }); }); // Pairing-code endpoint app.post("/api/pair", pairLimit, async (req, res) => { const phone = String(req.body?.phone || "").replace(/\D/g, ""); if (phone.length < 8 || phone.length > 15) { return res.status(400).json({ error: "Enter a valid international phone number with country code." }); } if (state.connected) { return res.status(409).json({ error: "WhatsApp is already connected." }); } if (state.pairing) { return res.status(429).json({ error: "Another pairing request is running. Please wait." }); } state.pairing = true; try { if (!state.sock) await start(); if (!state.sock || !state.sock.requestPairingCode) { return res.status(503).json({ error: "WhatsApp is not ready yet. Try again shortly." }); } const code = await state.sock.requestPairingCode(phone); res.json({ code, message: "WhatsApp → Linked devices → Link with phone number." }); } catch (err) { logger.error({ err }, "Pair-code request failed"); res.status(503).json({ error: "Pairing failed. Wait a moment and try again." }); } finally { state.pairing = false; } }); app.listen(PORT, () => { logger.info( { port: PORT, commands: commandCount }, "GHOST-MD server ready" ); });
